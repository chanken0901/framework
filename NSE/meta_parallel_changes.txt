! --- main.f90: use文 ---
use mod_slf_output, only : write_nse_conserved_slf, write_meta_json

! --- main.f90: MPI分割後にsimへ実行情報を入れる ---
sim%use_mpi = .true.
sim%rank    = my_rank
sim%nprocs  = nprocs
! OpenMP/CUDAは使用状況に合わせる
sim%use_openmp = .true.
! sim%backend = 'mpi_openmp'

! --- main.f90: js,je,ks,ke を決めた後、全rankで1回だけ呼ぶ ---
call write_meta_json(sim, js=js, je=je, ks=ks, ke=ke, use_cuda=.false.)
call mp_barrier

! --- mod_slf_output.f90: write_meta_json の宣言を変更 ---
subroutine write_meta_json(cfg, filename, primary_variables_note, js, je, ks, ke, use_cuda)
  use omp_lib, only : omp_get_max_threads
  type(simulation_config), intent(in) :: cfg
  character(len=*), intent(in), optional :: filename
  character(len=*), intent(in), optional :: primary_variables_note
  integer, intent(in), optional :: js, je, ks, ke
  logical, intent(in), optional :: use_cuda

  integer :: u, i, nvar, ios, ierr, r
  integer :: local_range(6)
  integer, allocatable :: all_ranges(:,:)
  integer :: omp_threads
  logical :: cuda_enabled

! --- local_range の意味 ---
! local_range = [i_start, i_end, j_start, j_end, k_start, k_end]
  local_range = [1, cfg%nx, 1, cfg%ny, 1, cfg%nz]
  if (present(js)) local_range(3) = js
  if (present(je)) local_range(4) = je
  if (present(ks)) local_range(5) = ks
  if (present(ke)) local_range(6) = ke

  if (my_rank == root) allocate(all_ranges(6,nprocs))
  call MPI_Gather(local_range, 6, MPI_INTEGER, all_ranges, 6, MPI_INTEGER, root, MPI_COMM_WORLD, ierr)

  cuda_enabled = .false.
  if (present(use_cuda)) cuda_enabled = use_cuda
  omp_threads = 1
  if (cfg%use_openmp) omp_threads = omp_get_max_threads()

! JSON出力部分の variables の後などに以下を追加:
  write(u,'(A)') '  "parallel": {'
  write(u,'(A,L1,A)') '    "mpi_enabled": ', cfg%use_mpi, ','
  write(u,'(A,I0,A)') '    "mpi_nprocs": ', nprocs, ','
  write(u,'(A,L1,A)') '    "openmp_enabled": ', cfg%use_openmp, ','
  write(u,'(A,I0,A)') '    "openmp_max_threads": ', omp_threads, ','
  write(u,'(A,L1,A)') '    "cuda_enabled": ', cuda_enabled, ','
  write(u,'(A)') '    "decomposition": "x-global_yz-block",'
  write(u,'(A)') '    "rank_ranges": ['
  do r = 0, nprocs-1
    write(u,'(A,I0,A,I0,A,I0,A,I0,A,I0,A,I0,A,I0,A)', advance='no') &
      '      {"rank": ', r, ', "i_start": ', all_ranges(1,r+1), ', "i_end": ', all_ranges(2,r+1), &
      ', "j_start": ', all_ranges(3,r+1), ', "j_end": ', all_ranges(4,r+1), &
      ', "k_start": ', all_ranges(5,r+1), ', "k_end": ', all_ranges(6,r+1), '}'
    if (r < nprocs-1) then
      write(u,'(A)') ','
    else
      write(u,'(A)') ''
    end if
  end do
  write(u,'(A)') '    ]'
  write(u,'(A)') '  }'

  if (allocated(all_ranges)) deallocate(all_ranges)
end subroutine write_meta_json
